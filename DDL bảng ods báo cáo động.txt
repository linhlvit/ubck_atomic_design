*** Bảng uploadfile
CREATE TABLE WEIGHT_CONFIG (
    ID                      NUMBER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    -- Lưu nguyên văn giá trị cột "Chỉ tiêu đầu ra" của file import, không resolve qua mã
    OUTPUT_INDICATOR_CODE   VARCHAR2(255)   NOT NULL,
    INPUT_INDICATOR_CODE    VARCHAR2(50)    NOT NULL,
    INPUT_INDICATOR_NAME    VARCHAR2(255),
    WEIGHT                  NUMBER          NOT NULL,
    EFFECTIVE_DATE          DATE            NOT NULL,
    END_DATE                DATE,
    -- 1 = đang active (END_DATE IS NULL), 0 = đã bị đóng bởi bản ghi mới hơn
    RECORD_STATUS           NUMBER(1)       DEFAULT 1 NOT NULL,
    CREATED_AT              DATE            DEFAULT SYSDATE NOT NULL,
    CREATEDBY               VARCHAR2(100),
    UPLOAD_HISTORY_ID       NUMBER,
	
    CONSTRAINT FK_WEIGHT_CONFIG_HISTORY
        FOREIGN KEY (UPLOAD_HISTORY_ID) REFERENCES upload_history(id)
);

CREATE TABLE THRESHOLD_CONFIG (
    ID                  NUMBER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    INDICATOR_CODE      VARCHAR2(255)   NOT NULL,
    INDICATOR_NAME      VARCHAR2(255),
    VALUE_FROM          NUMBER,
    VALUE_TO            NUMBER,
    STATUS              VARCHAR2(100),
    EFFECTIVE_DATE      DATE            NOT NULL,
    END_DATE            DATE,
    -- 1 = đang active (END_DATE IS NULL), 0 = đã bị đóng bởi bản ghi mới hơn
    RECORD_STATUS       NUMBER(1)       DEFAULT 1 NOT NULL,
    CREATED_AT          DATE            DEFAULT SYSDATE NOT NULL,
    CREATEDBY           VARCHAR2(100),
    UPLOAD_HISTORY_ID   NUMBER,
	
    CONSTRAINT FK_THRESHOLD_CONFIG_HISTORY
        FOREIGN KEY (UPLOAD_HISTORY_ID) REFERENCES upload_history(id)
);
